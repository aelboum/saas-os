from sqlalchemy import text
from saas_os.infra.db.config import get_migrations_database_config
from saas_os.infra.db.engine import build_engine
e = build_engine(get_migrations_database_config())
with e.connect() as c:
    print("stamp", c.execute(text("SELECT version_num FROM alembic_version_saas_os")).scalar_one())
    print("check", c.execute(text("SELECT conname, convalidated FROM pg_constraint WHERE conname = 'ck_outbox_events_external_started_pair'")).all())
    print("columns", c.execute(text("SELECT column_name FROM information_schema.columns WHERE table_schema = 'core' AND table_name = 'outbox_events' AND column_name LIKE 'external_started%' ORDER BY 1")).all())
    print("rows, rows with a marker", c.execute(text("SELECT count(*), count(*) FILTER (WHERE external_started_at IS NOT NULL OR external_started_claim IS NOT NULL) FROM core.outbox_events")).one())
e.dispose()
EOF
